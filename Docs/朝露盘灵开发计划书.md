## 一、项目概述

### 1.1 项目目标

将经典MC地图《盘灵古域》从1.8版本完整移植到1.21.1，使用纯插件方案替代原版命令方块系统，实现完整的RPG游戏体验。

### 1.2 原版规模参考

| 内容 | 数量 | 移植难度 |
|------|------|----------|
| 命令方块 | 20,000+ | ⭐⭐⭐⭐⭐ |
| 剧情文本 | 100,000字 | ⭐⭐⭐ |
| NPC | 700+ | ⭐⭐⭐⭐ |
| 种族 | 5个 | ⭐⭐ |
| 职业 | 3个 | ⭐⭐⭐ |
| 副本 | 4个主要 + 3个试炼 | ⭐⭐⭐⭐⭐ |
| 战役 | 3种 | ⭐⭐⭐⭐ |
| 装备体系 | 多阶级神器系统 | ⭐⭐⭐⭐ |

### 1.3 技术优势（相比命令方块）

- ✅ 更高的执行效率
- ✅ 更好的可维护性
- ✅ 支持数据库持久化
- ✅ 更灵活的扩展性
- ✅ 更好的多人服务器支持
- ✅ 版本升级更方便

### 3.5 战斗与技能系统

```kotlin
// Skill.kt
abstract class Skill(
    val id: String,
    val name: String,
    val description: String,
    val maxLevel: Int,
    val cooldown: Long,           // 毫秒
    val manaCost: Int,
    val requiredLevel: Int,
    val requiredRace: Race? = null,
    val requiredProfession: Profession? = null
) {
    abstract fun cast(caster: Player, target: LivingEntity?): Boolean
    abstract fun getEffect(level: Int): SkillEffect
    
    fun canCast(player: Player): CastResult {
        val dewPlayer = player.dewPlayer
        
        // 检查等级
        if (dewPlayer.level < requiredLevel) 
            return CastResult.LEVEL_TOO_LOW
            
        // 检查冷却
        if (isOnCooldown(player)) 
            return CastResult.ON_COOLDOWN
            
        // 检查蓝量
        if (dewPlayer.mana < manaCost) 
            return CastResult.NOT_ENOUGH_MANA
            
        return CastResult.SUCCESS
    }
}

// 伤害计算
object DamageCalculator {
    
    fun calculateDamage(
        attacker: DewPlayer,
        defender: DewPlayer,
        baseDamage: Double,
        damageType: DamageType
    ): Double {
        var damage = baseDamage
        
        // 攻击力加成
        damage += attacker.attributes[Attribute.ATTACK]
        
        // 暴击判定
        if (Random.nextDouble() < attacker.attributes[Attribute.CRIT_CHANCE]) {
            damage *= attacker.attributes[Attribute.CRIT_DAMAGE]
        }
        
        // 防御减伤
        val defense = defender.attributes[Attribute.DEFENSE]
        damage *= (100.0 / (100.0 + defense))
        
        // 种族克制
        damage *= getRaceModifier(attacker.race, defender.race)
        
        // Buff影响
        attacker.activeBuffs.forEach { damage = it.modifyOutgoingDamage(damage) }
        defender.activeBuffs.forEach { damage = it.modifyIncomingDamage(damage) }
        
        return damage.coerceAtLeast(1.0)
    }
}
```

### 3.6 副本系统

```kotlin
// Dungeon.kt
abstract class Dungeon(
    val id: String,
    val name: String,
    val minPlayers: Int,
    val maxPlayers: Int,
    val minLevel: Int,
    val timeLimit: Long,         // 秒
    val cooldown: Long           // 秒
) {
    abstract val phases: List<DungeonPhase>
    abstract val rewards: DungeonRewards
    
    abstract fun createInstance(team: List<Player>): DungeonInstance
}

// DungeonInstance.kt
class DungeonInstance(
    val dungeon: Dungeon,
    val team: List<Player>,
    val instanceWorld: World      // 副本独立世界
) {
    var currentPhase: Int = 0
    var startTime: Long = 0
    var status: InstanceStatus = InstanceStatus.WAITING
    
    fun start() {
        status = InstanceStatus.IN_PROGRESS
        startTime = System.currentTimeMillis()
        phases[0].start(this)
    }
    
    fun nextPhase() {
        currentPhase++
        if (currentPhase >= phases.size) {
            complete()
        } else {
            phases[currentPhase].start(this)
        }
    }
    
    fun complete() {
        status = InstanceStatus.COMPLETED
        distributeRewards()
        scheduleCleanup()
    }
    
    fun fail() {
        status = InstanceStatus.FAILED
        teleportPlayersOut()
        scheduleCleanup()
    }
}
```

# 推荐使用官方服务端逐版本升级

cd conversion/1.8.3
cp -r /path/to/original/world .
java -jar server.jar --nogui
# 等待转换完成后关闭

# 3. 复制到下一版本
cp -r world ../1.12.2/
cd ../1.12.2
java -jar server.jar --nogui
# 重复此过程...

# 关键版本节点：
